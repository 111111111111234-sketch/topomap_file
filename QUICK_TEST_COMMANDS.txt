python3 << 'EOF'
import sys
import numpy as np
from conftopo.agents.goat_agent_new import GoatAgent
from conftopo.core.instruction_graph import GoalNode, Relation

print("\n" + "="*70)
print("🚀 GoatAgent Pipeline Smoke Test (Real VLM)")
print("="*70)

try:
    print("\n[1/6] 创建 GoatAgent 实例...")
    agent = GoatAgent()
    print("      ✅ Agent 创建成功")
    
    print("\n[2/6] 设置导航目标...")
    goal = GoalNode(
        target_object="chair",
        attributes=["red"],
        room_prior=["dining room"],
        landmarks=["table"],
        relations=[Relation("near", "table")],
    )
    agent.set_new_goal(goal)
    print(f"      ✅ Goal 设置完成")
    print(f"         target_object: {goal.target_object}")
    print(f"         attributes: {goal.attributes}")
    print(f"         room_prior: {goal.room_prior}")
    
    print("\n[3/6] 观测环境 (Observe)...")
    obs = {
        "rgb": None,
        "rgb_embed": np.random.rand(512).astype("float32"),
        "position": np.array([0.0, 0.0, 0.0], dtype="float32"),
        "heading": 0.0,
    }
    agent.observe(obs)
    print(f"      ✅ 观测完成")
    print(f"         position: {obs['position']}")
    print(f"         heading: {obs['heading']}")
    
    print("\n[4/6] 更新记忆 (Update Memory)...")
    agent.update_memory()
    print("      ✅ 记忆更新完成")
    
    print("\n[5/6] 生成计划 (Plan)...")
    plan = agent.plan()
    print(f"      ✅ 计划生成完成")
    print(f"         plan type: {type(plan)}")
    print(f"         plan value: {plan}")
    
    print("\n[6/6] 执行动作 (Act)...")
    action_output = agent.act(plan)
    print(f"      ✅ 动作执行完成")
    print(f"         output: {action_output}")
    
    print("\n" + "="*70)
    print("📊 Agent 统计信息:")
    print("="*70)
    stats = agent.memory_stats()
    if stats:
        for key, value in stats.items():
            print(f"      {key}: {value}")
    else:
        print("      (无统计信息)")
    
    print("\n" + "="*70)
    print("✅✅✅ 所有测试通过！Pipeline 可以正常运行 ✅✅✅")
    print("="*70 + "\n")
    
except Exception as e:
    print(f"\n❌ 测试失败！")
    print(f"   错误类型: {type(e).__name__}")
    print(f"   错误信息: {e}\n")
    import traceback
    traceback.print_exc()
    sys.exit(1)

EOF
